Verilog Code for Serial In Serial Out (SISO) Shift Register

Verilog Code Print

module siso_4bit (
    input wire clk,          // 100 MHz clock
    input wire rst_n,        // Active-low reset
    input wire serial_in,
    output wire serial_out,
    output reg clk_div
);

    reg [3:0] shift_reg;
    reg [25:0] counter;

    // Divide 100 MHz clock to 1 Hz square wave
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            counter <= 26'd0;
            clk_div <= 1'b0;
        end
        else if (counter == 26'd49_999_999) begin
            counter <= 26'd0;
            clk_div <= ~clk_div;
        end
        else begin
            counter <= counter + 1'b1;
        end
    end

    // 4-bit SISO shift register
    always @(posedge clk_div or negedge rst_n) begin
        if (!rst_n)
            shift_reg <= 4'b0000;
        else
            shift_reg <= {shift_reg[2:0], serial_in};
    end

    assign serial_out = shift_reg[3];

endmodule






Verilog Test Bench code print

`timescale 1ns / 1ps

module siso_4bit_tb;

    reg clk;
    reg rst_n;
    reg serial_in;

    wire serial_out;
    wire clk_div;

    // Instantiate the SISO shift register
    siso_4bit DUT (
        .clk(clk),
        .rst_n(rst_n),
        .serial_in(serial_in),
        .serial_out(serial_out),
        .clk_div(clk_div)
    );

    // 10 ns clock period (100 MHz)
    initial begin
        clk = 1'b0;
        forever #5 clk = ~clk;
    end

    // Monitor signals
    initial begin
        $monitor(
            "Time=%0t | clk=%b | rst_n=%b | serial_in=%b | shift_reg=%b | clk_div=%b | serial_out=%b",
            $time, clk, rst_n, serial_in,
            DUT.shift_reg, clk_div, serial_out
        );
    end

    // Test sequence
    initial begin
        rst_n = 1'b0;
        serial_in = 1'b0;

        // Hold reset
        #20;
        rst_n = 1'b1;

        // Input sequence: 1, 0, 1, 1
        @(negedge clk) serial_in = 1'b1;
        @(negedge clk) serial_in = 1'b0;
        @(negedge clk) serial_in = 1'b1;
        @(negedge clk) serial_in = 1'b1;

        // Shift additional zeros
        @(negedge clk) serial_in = 1'b0;
        @(negedge clk) serial_in = 1'b0;
        @(negedge clk) serial_in = 1'b0;
        @(negedge clk) serial_in = 1'b0;

        #100;
        $finish;
    end

endmodule
