Verilog Code for Full Adder using two half adders: 
// Full adder built from two half adders + OR gate 
module half_adder (sum, carry, a, b);
 output sum, carry;
 input a, b;
 assign sum = a ^ b;
 assign carry = a & b;
endmodule 
module full_adder (sum, cout, a, b, cin);
 output sum, cout;
 input a, b, cin;
 wire s1, c1, c2;
 half_adder HA1 (s1, c1, a, b);
 half_adder HA2 (sum, c2, s1, cin);
 or OR1 (cout, c1, c2);
endmodule
