Develop Verilog code and suitable test bench to simulate a 4-bit shift register with two modes: 
Serial In Parallel Out (SIPO) and Parallel In Serial Out (PISO).
Verilog Code Print
module sipo_piso_reg (
 input wire clk, // 100 MHz input clock
 input wire rst,
 input wire mode, // 0 = SIPO, 1 = PISO
 input wire load, // Load parallel data in PISO mode
 input wire serial_in,
 input wire [3:0] parallel_in,
 output reg [3:0] parallel_out,
 output wire serial_out,
 output reg clk_div // 1 Hz divided clock
);
 reg [3:0] shift_reg;
 reg [25:0] counter;
 // Frequency divider: 100 MHz to 1 Hz
 always @(posedge clk or posedge rst) begin
 if (rst) begin
 counter <= 26'd0;
 clk_div <= 1'b0;
 end
 else if (counter == 26'd49_999_999) begin
 counter <= 26'd0;
 clk_div <= ~clk_div;
 end else begin
 counter <= counter + 1'b1;
 end
 end
 // SIPO / PISO register operating at 1 Hz
 always @(posedge clk_div or posedge rst) begin
 if (rst) begin
 shift_reg <= 4'b0000;
 parallel_out <= 4'b0000;
 end
 else begin
 if (mode == 1'b0) begin
 // SIPO operation
 shift_reg <= {shift_reg[2:0], serial_in};
 parallel_out <= {shift_reg[2:0], serial_in};
 end
 else begin
 // PISO operation
 if (load)
 shift_reg <= parallel_in;
 else
 shift_reg <= {1'b0, shift_reg[3:1]};
 end
 end
 end
 // Serial output during PISO mode
 assign serial_out = shift_reg[0]; endmodule
Verilog Test Bench code print
`timescale 1ns / 1ps
module sipo_piso_reg_tb;
 reg clk;
 reg rst;
 reg mode;
 reg load;
 reg serial_in;
 reg [3:0] parallel_in;
 wire [3:0] parallel_out;
 wire serial_out;
 wire clk_div;
 // Instantiate DUT
 sipo_piso_reg DUT (
 .clk(clk),
 .rst(rst),
 .mode(mode),
 .load(load),
 .serial_in(serial_in),
 .parallel_in(parallel_in),
 .parallel_out(parallel_out),
 .serial_out(serial_out),
 .clk_div(clk_div) );
 // 100 MHz input clock: 10 ns period
 always #5 clk = ~clk;
 initial begin
 // Initial values
 clk = 0;
 rst = 1;
 mode = 0;
 load = 0;
 serial_in = 0;
 parallel_in = 4'b0000;
 $monitor(
 "Time=%0t | rst=%b | mode=%b | load=%b | serial_in=%b | parallel_in=%b | parallel_out=%b 
| serial_out=%b | clk_div=%b",
 $time, rst, mode, load, serial_in,
 parallel_in, parallel_out, serial_out, clk_div
 );
 // Reset
 #12;
 rst = 0;
 // =====================================
 // TEST 1: SIPO (Serial-In Parallel-Out)
 // Send serial bits: 1, 0, 1, 1
 // =====================================
 mode = 0; load = 0;
 @(negedge clk_div); serial_in = 1;
 @(negedge clk_div); serial_in = 0;
 @(negedge clk_div); serial_in = 1;
 @(negedge clk_div); serial_in = 1;
 // Allow final SIPO result to settle
 @(posedge clk_div);
 #1;
 $display("SIPO result: parallel_out = %b", parallel_out);
 // =====================================
 // TEST 2: PISO (Parallel-In Serial-Out)
 // Load parallel data: 1011
 // =====================================
 @(negedge clk_div);
 mode = 1;
 parallel_in = 4'b1011;
 load = 1;
 @(posedge clk_div);
 #1;
 $display("PISO loaded: parallel_in = %b", parallel_in);
 // Disable load and shift out the bits
 @(negedge clk_div);
 load = 0;
 // Observe four serial output bits repeat (4) begin
 @(posedge clk_div);
 #1;
 $display("PISO serial_out = %b at time %0t",
 serial_out, $time);
 end
 #10;
 $finish;
 end
endmodule
