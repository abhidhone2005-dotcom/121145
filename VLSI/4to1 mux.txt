4:1 Multiplexer – Behavioral Modeling Code:
`timescale 1ns/1ps
module mux4_1 (w0, w1, w2, w3, S, f);
//4:1 Multiplexer - Behavioral Modeling
input w0, w1, w2, w3;
input [1:0] S;
output reg f;
always @(*)
if (S == 2'b00)
f = w0;
else if (S == 2'b01)
f = w1;
else if (S == 2'b10)
f = w2;
else
f = w3;
endmodule



4:1 Multiplexer – Dataflow Modeling
module mux4_1 (w0, w1, w2, w3, S, f);
input w0, w1, w2, w3;
input [1:0] S;
output f;
assign f = S[1] ? (S[0] ? w3 : w2) : (S[0] ? w1 : w0);
endmodule



4:1 Multiplexer Using Hierarchy
module mux2_1 (y, i0, i1, sel);
output y;
input i0, i1, sel;
assign y = sel ? i1 : i0;
endmodule
module mux4_1 (f, w0, w1, w2, w3, s1, s0);
output f;
input w0, w1, w2, w3;
input s1, s0;
wire y1, y2;
// Instantiate 2:1 MUX modules
mux2to1 M1 (y1, w0, w1, s0);
mux2to1 M2 (y2, w2, w3, s0);
mux2to1 M3 (f, y1, y2, s1);
endmodule

8:1 Multiplexer – Hierarchical Modeling Code:
// 2:1 Multiplexer
module mux2to1(y, i0, i1, sel);
input i0, i1, sel;
output y;
assign y = sel ? i1 : i0;
endmodule
// 4:1 Multiplexer using 2:1 MUX
module mux4to1(f, w0, w1, w2, w3, s1, s0);
input w0, w1, w2, w3;
input s1, s0;
output f;
wire y1, y2;
mux2to1 M1(y1, w0, w1, s0);
mux2to1 M2(y2, w2, w3, s0);
mux2to1 M3(f, y1, y2, s1);
endmodule
// 8:1 Multiplexer using 4:1 MUX and 2:1 MUX
module mux8_1(
 output f,
 input w0, w1, w2, w3,
 input w4, w5, w6, w7,
 input s2, s1, s0
);
wire y1, y2;
mux4to1 M4(y1, w0, w1, w2, w3, s1, s0);
mux4to1 M5(y2, w4, w5, w6, w7, s1, s0);
mux2to1 M6(f, y1, y2, s2);
endmodule
