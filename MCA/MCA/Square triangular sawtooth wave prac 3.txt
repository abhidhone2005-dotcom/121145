TRIANGULARWAVE -
#include <reg51.h>	 
// Function prototype for delay void delay();
void main()
{
	while(1)	//Infinite loop
{
int i, j;
// Increase value from 0 to 255
for(i= 0;i<= 255;i++)
{
P1=i; //SendincreasingvaluetoPort 1 delay(); //Smalldelay
}
// Decrease value from 255 to 0
for(j = 255; j >= 0; j--)
{
P1=j; //SenddecreasingvaluetoPort 1 delay(); //Smalldelay
}
}
}
//Timer0delay function
void delay()
{
TMOD=0x01;
TL0=0x1A; TH0=0xFF;
TR0= 1; while(TF0==0);
TR0= 0; 
TF0= 0; 
} 	//Timer0,Mode1(16-bit timer) 
//LoadlowerbyteofTimer 0
//LoadhigherbyteofTimer 0 
//StartTimer0
//Waitfortimeroverflow 
//StopTimer0
//Clearoverflow flag
